extends Node
func _ready() -> void:
	var out := OS.get_cmdline_user_args()[0]
	var bg := ColorRect.new()
	bg.color = Color(0.2, 0.2, 0.24)
	bg.size = Vector2(1280, 720)
	add_child(bg)
	var lab := Label.new()
	lab.text = OS.get_cmdline_user_args()[1]
	lab.add_theme_font_size_override("font_size", 26)
	lab.position = Vector2(20, 4)
	add_child(lab)
	var scene: PackedScene = load("res://scenes/player/stroller.tscn")
	var dirs := [Vector2.UP, Vector2.RIGHT, Vector2.DOWN]
	for row in 2:
		for col in 2:
			var s: Stroller = scene.instantiate()
			s.carrying = true
			s.is_male = row == 1
			s.set_physics_process(false)
			add_child(s)
			s.position = Vector2(330 + col * 460, 330 + row * 360)
			s.get_node("Camera2D").enabled = false
			s.facing = dirs[1]
			s._view_direction = s._nearest_view_direction()
			var b: Baby = s.get_node("Baby")
			b.set_physics_process(false)
			b.state = GameEnums.BabyState.ASLEEP
			s.warn(Stroller.Alert.NOW if col % 2 == 1 else Stroller.Alert.SOON, 99.0, &"x")
			s.scale = Vector2(3.5, 3.5)
			s.queue_redraw()
	for i in 4:
		await get_tree().process_frame
	await RenderingServer.frame_post_draw
	get_viewport().get_texture().get_image().save_png(out)
	get_tree().quit()
